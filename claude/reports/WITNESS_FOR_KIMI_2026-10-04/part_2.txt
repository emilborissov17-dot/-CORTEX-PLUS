116      subs_of: dict = {}
117      for r in ix.get("subcategory", []):

122          for g in ix.get("gap", []):

124                  cats = {r[0] for r in ix.get("serves", []) if len(r) == 2 and r[1] == sg}
125                  if not any(s in obs_subs for c in cats for s in subs_of.get(c, ())):
126                      out.append(["uncovered", sg])
