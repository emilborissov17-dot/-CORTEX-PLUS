# -*- coding: utf-8 -*-
"""M-BN1: the pre-registered measurement for need BN-6b3b68e0b0 (claude/PREREG_DRAFT_2026-10-10_BN-6b3b68e0b0.md).
READ-ONLY: reads memory/vertical_ledger.jsonl, memory/statements.jsonl, memory/openclaw_pages/; writes only
its own report to the path given. Runs no model. Usage:
  python measure.py <repo> <start_utc> <report.json>
start_utc = the moment the protocol was sealed. Takes before it only seed the set of URLs already seen."""
import bisect, json, pathlib, random, re, sys

PARENT, CHILD = "BN-6b3b68e0b0", "BN-0e1657dcc6"
LINEAGE = {PARENT: "parent", CHILD: "child"}
END_AFTER = 6            # consecutive takes with no new URL (Perplexity 85C, decided)
SAMPLE_ALL_UP_TO, SAMPLE_N, SEED = 20, 20, 20261010   # PENDING Emil's decision (prereg §8)
NUM = re.compile(r"(?<![\w.])(?!(?:19|20)\d\d(?!\d))\d[\d,.]*")
ENV = re.compile(r"\b(air|pollut|PM2\.5|PM10|ozone|smog|haze|visibility|water|drought|flood|heat|temperatur|wildfire|"
                 r"fire|forest|deforest|soil|biodivers|species|ocean|sea|coral|emission|CO2|carbon|climate|glacier|"
                 r"rain|precipit|storm|cyclone)", re.I)


def places():
    import pycountry
    n = set()
    for c in pycountry.countries:
        for a in ("name", "common_name", "official_name"):
            if getattr(c, a, None):
                n.add(getattr(c, a))
    for s in pycountry.subdivisions:
        n.add(s.name)
    return re.compile(r"\b(" + "|".join(sorted(map(re.escape, n), key=len, reverse=True)) + r")\b")


def main():
    repo, start, out = pathlib.Path(sys.argv[1]), sys.argv[2], pathlib.Path(sys.argv[3])
    PL = places()
    seen, takes, raw = set(), [], []
    cur = None
    for l in open(repo / "memory/vertical_ledger.jsonl", encoding="utf-8", errors="replace"):
        try:
            r = json.loads(l)
        except ValueError:
            continue
        if r.get("need_id") not in LINEAGE:
            continue
        before = r.get("ts", "") < start
        if r.get("event") == "TAKEN" and not before:
            cur = {"ts": r["ts"], "need": LINEAGE[r["need_id"]], "new_urls": 0}
            takes.append(cur)
        if r.get("event") == "FETCHED":
            u = r.get("url")
            if u not in seen and not before and cur is not None:
                cur["new_urls"] += 1
            seen.add(u)
        if not before:
            raw.append(r)
    run, end_ts = 0, None
    for t in takes:
        run = 0 if t["new_urls"] else run + 1
        if run >= END_AFTER:
            end_ts = t["ts"]; break
    items = []
    for l in open(repo / "memory/statements.jsonl", encoding="utf-8", errors="replace"):
        if PARENT not in l and CHILD not in l:
            continue
        try:
            s = json.loads(l)
        except ValueError:
            continue
        if s.get("need_id") not in LINEAGE or (s.get("ingested_at") or "")[:19] + "Z" < start:
            continue
        if end_ts and (s.get("ingested_at") or "")[:19] + "Z" > end_ts:
            continue
        t = s.get("sentence") or ""
        a, b, c, d = bool(NUM.search(t)), "2025" in t, bool(PL.search(t)), bool(ENV.search(t))
        if not (a and b and c and d):
            continue
        page = repo / "memory/openclaw_pages" / f"{s.get('page_sha256')}.json"
        check1 = False
        if page.exists():
            txt = json.loads(page.read_text(encoding="utf-8")).get("text") or ""
            check1 = all(x in txt for x in NUM.findall(t)[:1] + ["2025"])
        items.append({"id": s.get("id"), "need": LINEAGE[s["need_id"]], "region": s.get("region"), "url": s.get("url"),
                      "page_sha256": s.get("page_sha256"), "sentence": t, "place": PL.search(t).group(0),
                      "check1_same_path": check1})
    uniq = {}
    for it in items:
        uniq.setdefault(it["sentence"].strip(), it)
    cand = sorted(uniq.values(), key=lambda x: x["id"])
    sample = cand if len(cand) <= SAMPLE_ALL_UP_TO else random.Random(SEED).sample(cand, SAMPLE_N)
    rep = {"start": start, "window_end": end_ts, "window_closed": bool(end_ts), "takes": takes,
           "candidates": len(cand), "items": cand, "sample_for_emil": [x["id"] for x in sample],
           "raw_ledger_rows": raw, "note": "candidates are heuristic numeric-fact candidates, not validated measurements"}
    out.write_text(json.dumps(rep, ensure_ascii=False, indent=1), encoding="utf-8")
    print(f"takes {len(takes)} window_closed {bool(end_ts)} candidates {len(cand)} sample {len(sample)}")


if __name__ == "__main__":
    main()
